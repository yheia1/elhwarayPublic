    function updatePermissions() {
      const username = document.getElementById("username").value;
      const checkboxes = document.querySelectorAll("input[type=checkbox]:checked");
      const permissions = Array.from(checkboxes).map(cb => cb.value);

      fetch("/update-permissions", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ targetUsername: username, permissions })
      })
      .then(res => res.json())
      .then(data => {
        const result = document.getElementById("result");
        result.style.color = data.success ? 'green' : 'red';
        result.innerText = data.success ? "?  „  ÕœÌÀ «·’·«ÕÌ«  »‰Ã«Õ" : "? «·„” Œœ„ €Ì— „ÊÃÊœ";

    function loadAdmins() {
      fetch("/admins")
        .then(res => res.json())
        .then(data => {
          const tbody = document.querySelector("#adminsTable tbody");
          tbody.innerHTML = "";
          data.forEach(admin => {
            const row = `<tr><td>${admin.username}</td><td>${admin.permissions.join(", ")}</td></tr>`;
            tbody.innerHTML += row;
          });
        });
    }

    function loadUserInfo() {
      fetch("/user-info")
        .then(res => res.json())
        .then(data => {
          document.getElementById("ip").innerText = data.ip;
          document.getElementById("agent").innerText = data.userAgent;
        });
    }
